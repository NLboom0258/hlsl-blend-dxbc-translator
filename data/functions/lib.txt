float3 GetToonNoL(float Smoothness, float3 Normal, float3 LightDir)
{
    float ToonNoL = dot(normalize(Normal), normalize(-LightDir));
    ToonNoL = smoothstep(-Smoothness, Smoothness, ToonNoL);
    return ToonNoL;
}

void LightDirBlend(inout float3 LightDirSum, inout float TotalWeight, float VisibilityFactor, float BaseWeight, float3 PointLightDir)
{
    // 最终权重 = 基础权重 × 朝向调制因子
    float weight = BaseWeight * VisibilityFactor;
    
    // 累积光照向量
    LightDirSum += PointLightDir * weight;
    // 累积权重
    TotalWeight += TotalWeight + weight;
}

float3 ToonLightSpeculer(float3 ilmMap, float3 LightColor, float intensity, float SpeculerHardness, float3 LightDir, float3 Normal, float3 CameraDirVector, float3 CameraVector)
{    
    // 半角向量
    float HalfVector = saturate(dot(normalize(normalize(-LightDir) * 2 + CameraVector * 7), Normal));
    
    // fresnel_spec
    float RimSpecSize = 1.0;
    float eVecDif = dot(Normal, CameraDirVector);
    float fresnel_spec = 1 - eVecDif;
    fresnel_spec = pow(fresnel_spec, 5);
    fresnel_spec *= saturate(ilmMap.x - 0.75) * 8 * RimSpecSize;
    
    // Specular
    float Specular = clamp(pow(abs(HalfVector) + fresnel_spec, 4), 0.0, 0.99);
    
    // Speculer_Area
    float FrontSpecSize = 1.0;
    //float Speculer_Area = step(1 - ilmMap.z, Specular * FrontSpecSize); // 纯硬高光，没有软过渡，更不受金属度控制
    
    float Speculer_Area = saturate(1 - ilmMap.y * SpeculerHardness);
    //float Speculer_Area = saturate(1 - 1.0 * SpeculerHardness);
    Speculer_Area = smoothstep(1 - ilmMap.z - Speculer_Area, 1 - ilmMap.z + Speculer_Area, Specular * FrontSpecSize);
    
    // Speculer_Color
    float3 Speculer_Color = saturate(LightColor * intensity * ilmMap.x) * Speculer_Area;
    
    return Speculer_Color;
}

float3 ToonRimLight(float3 LightColor, float3 Normal, float3 CameraVector, float3 LightDir)
{
    float eVecDif = dot(Normal, CameraVector);
    float NdotL = GetToonNoL(1, Normal, LightDir);
    
    float eVecDif_rim = smoothstep(0.5, 1.0, NdotL) * 0.5 + 1;
    
    // 混合内边缘光和综合边缘光(混合暗部固有边缘光加上按视角和光照方向点乘结果控制显示的等宽边缘光)
    float3 RimLight = smoothstep(0.65, 0.6, eVecDif_rim - pow(1 - eVecDif, 2));
    float InnerRimMask = (1 - smoothstep(0.5, 0.75, NdotL)) * RimLight;
    RimLight = smoothstep(0.65, 0.6, 1 - pow(1 - eVecDif, 2));
    float OverallRimMask = dot(CameraVector, LightDir) * RimLight;
    
    RimLight = max(OverallRimMask, InnerRimMask);
    RimLight *= LightColor;
    
    return RimLight;
}

float3 GetMatCap(Texture2D MatCapTex, float4 MatcapOffset, SamplerState Sampler, float3 CameraVector, float3 CameraUpDirVector, float3 Normal)
{
    float2 MatcapUV;
    float3 CameraRightVector = normalize(cross(CameraVector.xzy, CameraUpDirVector.xzy));
    float3 CameraUpVector = cross(CameraVector.xzy, CameraRightVector);
    MatcapUV.x = dot(CameraRightVector.xyz, Normal.xyz);
    MatcapUV.y = dot(CameraUpVector, Normal);
    
    MatcapUV *= MatcapOffset.xy;
    MatcapUV += MatcapOffset.zw;
    
    float3 MatCap = MatCapTex.Sample(Sampler, MatcapUV);
    
    return MatCap;
}

float2 OffsetShadowUV(float2 originalUV, float sampleIndex, float totalSamples, float blurRadius, float2 TexSize)
{
    float GoldenAngle = 2.39996323;

    // 黄金角螺旋生成归一化偏移量 [0,1]
    float radius = sqrt((sampleIndex + 1.0) / totalSamples);
    float theta = GoldenAngle * sampleIndex;
    float2 offset;
    sincos(theta, offset.y, offset.x);
    offset *= radius;

    // 从纹素偏移转换为UV偏移，并应用模糊半径
    offset *= blurRadius / TexSize;

    // 偏移原UV并限制在[0,1]内
    float2 sampleUV = originalUV + offset;
    sampleUV = saturate(sampleUV);
    return sampleUV;
}

float2 GetGoldenSpiralOffset(float sampleIndex, float totalSamples, float radiusScale)
{
    float GoldenAngle = 2.39996323;
    float radius = sqrt((sampleIndex + 1.0) / totalSamples) * radiusScale; // radiusScale 可调模糊强度
    float theta = GoldenAngle * sampleIndex;
    float2 offset;
    sincos(theta, offset.y, offset.x);
    offset *= radius;
    return offset;
}
